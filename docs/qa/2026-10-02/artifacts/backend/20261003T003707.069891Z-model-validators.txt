921: def _read_header_row(reader: csv.DictReader) -> tuple[list[str], Optional[str]]:
922:     """The header row as the parser reads it, and why it cannot be a real one (``None`` if it can).
923: 
924:     An unterminated quote in the header row makes the reader take everything
925:     up to the next closing quote, often the rest of the file, as one header.
926:     Such a row spans more than one physical line; ``malformed_header_reason``
927:     also refuses line breaks and over-long headers. The reason never quotes
928:     the header, which may hold row data.
929:     """
930:     headers = [header or "" for header in (reader.fieldnames or [])]
931:     if reader.line_num > 1:
932:         return headers, f"the header row spans {reader.line_num} lines"
933:     return headers, malformed_header_reason(headers)

967: def _file_cte_type_for_jev(text: str, cte_type: Optional[str]) -> tuple[Optional[str], Optional[str]]:
968:     """``(file_cte_type to send, reason a declared one was withheld)``.
969: 
970:     Jev is told "every row in the file records <cte_type>" only when no row
971:     can contradict it. The declared ``cte_type`` is only the parser's
972:     fallback: ``_detect_row_cte_type`` reads a row's own CTE first, and that
973:     can come from two places.
974: 
975:     * A per-row CTE column the parser reads (``_CTE_TYPE_COLUMNS`` after
976:       ``normalize_csv_header``): withheld on the header alone.
977:     * An unrecognised column that Jev may map onto ``cte_type`` (``Activity``,
978:       ``Event Kind``, ...): withheld when any of its values, normalised the
979:       way ``_detect_row_cte_type`` normalises them, resolves through
980:       ``_CTE_TYPE_ALIASES`` to a different CTE.
981: 
982:     It reads exactly the rows ``parse_csv_rows`` reads (the first
983:     :data:`MAX_CSV_ROWS` the same ``csv.DictReader`` yields) and skips comment
984:     rows exactly as it does; a value in a row the parser discards cannot
985:     become a row's CTE. Recognised columns are never sent to Jev, so they
986:     cannot be mapped and are not scanned. Headers are normalised once per
987:     file, not per cell, and the scan stops at the first contradiction. It
988:     errs on withholding, which only costs Jev a hint: a single-letter alias
989:     (``S``, ``R``, ``H``, ...) in any unrecognised column counts, and so does
990:     a file whose rows cannot be read.
991: 
992:     Synchronous CPU work: the route runs it in the threadpool
993:     (``_read_upload_for_jev``).
994:     """
995:     if not cte_type:
996:         return None, None
997:     reader = csv.DictReader(io.StringIO(text))
998:     try:
999:         headers, malformed = _read_header_row(reader)
1000:         if malformed:
1001:             # Never scan, and never name, headers that may be row data.
1002:             return None, "the header row is malformed"
1003:         column = _per_row_cte_column(headers)
1004:         if column is not None:
1005:             return None, f"file has a per-row CTE column: {column}"
1006:         # The keys every DictReader row carries, normalised once.
1007:         raw_keys = list(dict.fromkeys(reader.fieldnames or []))
1008:         normalized = {key: normalize_csv_header(key) for key in raw_keys}
1009:         # Every raw column whose normalised form is unrecognised, including
1010:         # both of two headers that normalise alike: the parser keeps only the
1011:         # later one, so scanning both only errs on withholding.
1012:         scan_keys = [key for key in raw_keys if normalized[key] and normalized[key] not in _RECOGNIZED_COLUMNS]
1013:         if not scan_keys:
1014:             return cte_type, None
1015:         # parse_csv_rows tests the first value of {normalize(key): value}: the
1016:         # first key's normalised form, holding the value of the last raw key
1017:         # that normalises the same way.
1018:         comment_key = [key for key in raw_keys if normalized[key] == normalized[raw_keys[0]]][-1]
1019:         for raw_row in itertools.islice(reader, MAX_CSV_ROWS):
1020:             first = raw_row.get(comment_key)
1021:             if isinstance(first, str) and first.strip().startswith("#"):
1022:                 continue
1023:             for key in scan_keys:
1024:                 value = raw_row.get(key)
1025:                 if not value or not isinstance(value, str):
1026:                     continue
1027:                 row_cte = _CTE_TYPE_ALIASES.get(value.strip().lower().replace(" ", "_"))
1028:                 if row_cte and row_cte != cte_type:
1029:                     return None, f"column {key.strip()} holds per-row CTE values"
1030:     except csv.Error:
1031:         return None, "rows could not be read to rule out per-row CTE values"
1032:     return cte_type, None
