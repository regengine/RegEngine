27:   const rows: string[][] = [];
28:   let row: string[] = [], cell = '', quoted = false, closed = false;
29:   const finishCell = () => { row.push(cell.trim()); cell = ''; closed = false; };
30:   const finishRow = () => { finishCell(); if (row.some(value => value !== '')) rows.push(row); row = []; };

32:   for (let i = 0; i < text.length; i++) {
33:     const char = text[i];
34:     if (quoted) {
35:       if (char === '"') {
36:         if (text[i + 1] === '"') { cell += '"'; i++; }
37:         else { quoted = false; closed = true; }
38:       } else { cell += char; }
39:     } else if (char === ',') { finishCell(); }
40:     else if (char === '\r' || char === '\n') {
41:       finishRow(); if (char === '\r' && text[i + 1] === '\n') i++;
42:     } else if (char === '"') {
43:       if (cell || closed) throw new Error('Invalid quote in CSV field.');
44:       quoted = true;
45:     } else {
46:       if (closed) throw new Error('Unexpected text after a quoted CSV field.');
47:       cell += char;
48:     }
49:   }
50:   if (quoted) throw new Error('Unclosed quoted CSV field.');
51:   if (cell || row.length || closed) finishRow();

53:     const headers = rows[0].map(columnName);
54:     if (headers.some(header => !header) || new Set(headers).size !== headers.length) throw new Error('CSV headers must be non-empty and unique, including column aliases.');
