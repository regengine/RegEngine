133: async function deleteBySyncedAndCutoff<T extends OfflineRecord>(
134:     table: OfflineRecordTable<T>,
135:     synced: number,
136:     cutoff: number
137: ) {
138:     if (!Number.isFinite(cutoff)) return 0;
139:     return table

141:         .filter((record) => record.timestamp < cutoff)
142:         .delete();

145: /**
146:  * Enforce client-side retention for offline field capture data.
147:  *
148:  * Synced records are short lived cache. Unsynced records are preserved by
149:  * default because they are the durable source of truth until the backend
150:  * acknowledges persistence. Callers must opt in before deleting stale
151:  * unsynced records.
152:  */
153: export async function cleanupOfflineRecords(
154:     options: OfflineRetentionOptions = {},
155:     database: OfflineDatabase = db
156: ) {
157:     const now = options.now ?? Date.now();
158:     const syncedMaxAgeMs = options.syncedMaxAgeMs ?? OFFLINE_SYNC_RETENTION.syncedMaxAgeMs;
159:     const deleteUnsyncedRecords = options.deleteUnsyncedRecords === true;
160:     const unsyncedMaxAgeMs = options.unsyncedMaxAgeMs ?? OFFLINE_SYNC_RETENTION.unsyncedMaxAgeMs;
161:     const syncedCutoff = now - syncedMaxAgeMs;
162:     const unsyncedCutoff = now - unsyncedMaxAgeMs;
163: 
164:     const [
165:         deletedSyncedScans,
166:         deletedSyncedPhotos,
167:         deletedStaleUnsyncedScans,
168:         deletedStaleUnsyncedPhotos,
169:     ] = await Promise.all([
170:         deleteBySyncedAndCutoff(database.scans, 1, syncedCutoff),
171:         deleteBySyncedAndCutoff(database.photos, 1, syncedCutoff),
172:         deleteUnsyncedRecords ? deleteBySyncedAndCutoff(database.scans, 0, unsyncedCutoff) : Promise.resolve(0),
173:         deleteUnsyncedRecords ? deleteBySyncedAndCutoff(database.photos, 0, unsyncedCutoff) : Promise.resolve(0),
174:     ]);
175: 
176:     return {
177:         deletedSyncedScans,
178:         deletedSyncedPhotos,
179:         deletedStaleUnsyncedScans,
180:         deletedStaleUnsyncedPhotos,
181:     };
182: }

184: /**
185:  * Remove synced records older than `maxAgeMs` (default 7 days).
186:  * Prevents IndexedDB from growing indefinitely on mobile devices.
187:  */
