   356	def _get_active_portal_link(portal_id: str) -> dict:
   357	    # Try DB first, fall back to in-memory
   358	    link = _db_get_portal_link(portal_id)
   359	    if link is None and _allow_memory_fallback():
   360	        link = _portal_links.get(portal_id)
   361	    if not link:
   362	        raise HTTPException(status_code=404, detail="Portal link not found or expired")
   363	
   364	    expires_at_raw = link.get("expires_at")
   365	    if not expires_at_raw:
   366	        raise HTTPException(status_code=404, detail="Portal link not found or expired")
   367	
   368	    try:
   369	        expires_at = datetime.fromisoformat(str(expires_at_raw).replace("Z", "+00:00"))
   370	    except ValueError:
   371	        raise HTTPException(status_code=400, detail="Portal link has invalid expiry metadata")
   372	
   373	    if expires_at <= datetime.now(timezone.utc):
   374	        _portal_links.pop(portal_id, None)
   375	        _db_update_portal_link_status(portal_id, "expired", link["tenant_id"])
   376	        raise HTTPException(status_code=404, detail="Portal link not found or expired")
   377	
   378	    return link
