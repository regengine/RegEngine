    57	def get_current_active_session_context(
    58	    request: Request,
    59	    db: Session = Depends(get_session),
    60	    crypto_service: AuthCryptoService = Depends(get_auth_crypto_service),
    61	) -> ActiveSessionContext:
    62	    """Authenticate the current user and DB session from the opaque cookie."""
    63	    session_token = request.cookies.get(AUTH_SESSION_COOKIE)
    64	    if not session_token:
    65	        raise _unauthenticated()
    66	
    67	    try:
    68	        token_hash = crypto_service.hash_session_token(session_token)
    69	    except ValueError as exc:
    70	        raise _unauthenticated() from exc
    71	
    72	    now = datetime.now(timezone.utc)
    73	    stmt = (
    74	        select(UserModel, SessionModel)
    75	        .join(SessionModel, SessionModel.user_id == UserModel.id)
    76	        .where(
    77	            SessionModel.token_hash == token_hash,
    78	            SessionModel.is_revoked.is_(False),
    79	            SessionModel.revoked_at.is_(None),
    80	            SessionModel.expires_at > now,
    81	            UserModel.global_status == "active",
    82	            UserModel.status == "active",
    83	        )
    84	        .limit(1)
    85	    )
    86	
    87	    try:
    88	        row = db.execute(stmt).one_or_none()
    89	    except Exception as exc:
    90	        logger.error("session_lookup_failed", error=str(exc))
    91	        raise HTTPException(
    92	            status_code=status.HTTP_503_SERVICE_UNAVAILABLE,
    93	            detail="Authentication service temporarily unavailable.",
    94	        ) from exc
    95	
    96	    if row is None:
    97	        raise _unauthenticated()
    98	
    99	    user, session = row
   100	    return ActiveSessionContext(user=user, session=session)

    84	class PrivilegedAuthThrottle:
    85	    """Shared invalid-authentication counter with a local-development fallback.
    86	
    87	    Restricted staging/cloud and production never fall back to process memory.
    88	    If Redis is missing or an operation fails, privileged authentication
    89	    returns 503 rather than serving a request with an unverifiable
    90	    cross-replica failure budget.
    91	    """
    92	
    93	    def __init__(self, redis_client: object = _REDIS_CLIENT_UNSET) -> None:
    94	        self._redis = None if redis_client is _REDIS_CLIENT_UNSET else redis_client
    95	        self._redis_initialized = redis_client is not _REDIS_CLIENT_UNSET
    96	        self._memory: dict[str, list[float]] = defaultdict(list)
    97	        self._memory_lock = threading.Lock()
    98	        self._degraded_logged = False
    99	
   100	    @staticmethod
   101	    def _redis_url() -> str | None:
   102	        return os.getenv("REGENGINE_REDIS_URL") or os.getenv("REDIS_URL")
   103	
   104	    async def _get_redis(self):
   105	        if self._redis_initialized:
   106	            return self._redis
   107	
   108	        self._redis_initialized = True
   109	        redis_url = self._redis_url()
   110	        if not redis_url:
   111	            return None
   112	
   113	        try:
   114	            import redis.asyncio as redis
   115	
   116	            self._redis = redis.from_url(
   117	                redis_url,
   118	                socket_connect_timeout=1,
   119	                socket_timeout=1,
   120	            )
   121	        except Exception as exc:  # pragma: no cover - import/config environment dependent
   122	            self._backend_failure("initialize", exc)
   123	        return self._redis
   124	
   125	    @staticmethod
   126	    def _memory_fallback_allowed() -> bool:
   127	        environment = get_environment()
   128	        cloud_deployment = any(os.getenv(name) for name in CLOUD_ENV_VARS)
   129	        return not cloud_deployment and environment in _LOCAL_THROTTLE_ENVIRONMENTS
   130	
   131	    def _backend_failure(self, operation: str, exc: Exception | None = None) -> None:
   132	        fail_closed = not self._memory_fallback_allowed()
   133	        if not self._degraded_logged:
   134	            self._degraded_logged = True
   135	            log = logger.error if fail_closed else logger.warning
   136	            log(
   137	                "privileged_auth_throttle_unavailable",
   138	                operation=operation,
   139	                error_class=type(exc).__name__ if exc else "Unavailable",
   140	                fail_mode="closed" if fail_closed else "local_memory",
   141	            )
   142	        if fail_closed:
   143	            raise _authentication_service_unavailable() from exc

   145	    @staticmethod
   146	    def _key(subject: str) -> str:
   147	        # Avoid retaining a source IP verbatim in the Redis keyspace.
   148	        digest = hashlib.sha256(subject.encode("utf-8")).hexdigest()
   149	        return f"privileged_auth:failures:{digest}"

   151	    def _memory_is_limited(self, subject: str) -> bool:
   152	        now = time.monotonic()
   153	        with self._memory_lock:
   154	            recent = [
   155	                timestamp
   156	                for timestamp in self._memory[subject]
   157	                if now - timestamp < PRIVILEGED_AUTH_WINDOW_SECONDS
   158	            ]
   159	            self._memory[subject] = recent
   160	            return len(recent) >= PRIVILEGED_AUTH_MAX_FAILURES
   161	
   162	    def _memory_record_failure(self, subject: str) -> int:
   163	        now = time.monotonic()
   164	        with self._memory_lock:
   165	            recent = [
   166	                timestamp
   167	                for timestamp in self._memory[subject]
   168	                if now - timestamp < PRIVILEGED_AUTH_WINDOW_SECONDS
   169	            ]
   170	            recent.append(now)
   171	            self._memory[subject] = recent[-(PRIVILEGED_AUTH_MAX_FAILURES * 2) :]
   172	            return len(recent)

   174	    async def is_limited(self, subject: str) -> bool:
   175	        client = await self._get_redis()
   176	        if client is None:
   177	            self._backend_failure("read")
   178	            return self._memory_is_limited(subject)
   179	
   180	        try:
   181	            raw_count = await client.get(self._key(subject))
   182	            if raw_count is None:
   183	                self._degraded_logged = False
   184	                return False
   185	            count = int(raw_count)
   186	            self._degraded_logged = False
   187	            return count >= PRIVILEGED_AUTH_MAX_FAILURES
   188	        except HTTPException:
   189	            raise
   190	        except Exception as exc:
   191	            self._backend_failure("read", exc)
   192	            return self._memory_is_limited(subject)

   195	        client = await self._get_redis()
   196	        if client is None:
   197	            self._backend_failure("increment")
   198	            return self._memory_record_failure(subject)
