import { test, expect } from '@playwright/test';
import { isolateRegressionBrowser, writeRegressionEvidence } from './fe-regression-support';

test('FE-001 preserves complete GS1 lot identifiers in the real scanner form', async ({ context, page, baseURL }, testInfo) => {
  await isolateRegressionBrowser(context, baseURL!);
  const errors: string[] = [];
  page.on('pageerror', error => errors.push(error.message));
  const response = await page.goto('/tools/scan');
  expect(response?.status()).toBe(200);
  await expect(page.getByRole('heading', { name: 'RegEngine • Scan & Ingest', exact: true })).toBeVisible();
  await expect(page.getByRole('heading', { name: 'Save your GS1 scan result', exact: true })).toBeVisible();
  const input = page.getByLabel('Paste or type barcode string…');
  const lotField = page.getByLabel('Lot Code', { exact: true });
  const enterBarcode = async (barcode: string) => {
    // Firefox's native fill path strips ASCII GS; set the actual DOM input
    // with the native setter and notify React, exactly retaining scanner bytes.
    await input.evaluate((element, value) => {
      Object.getOwnPropertyDescriptor(HTMLInputElement.prototype, 'value')!.set!.call(element, value);
      element.dispatchEvent(new Event('input', { bubbles: true }));
    }, barcode);
    expect(await input.inputValue()).toBe(barcode);
  };

  // Both a simple AI lot and an encoded Digital Link must work first.
  for (const [barcode, lot] of [
    ['010950600013435210ALPHA\x1d17260930', 'ALPHA'],
    ['/01/09506000134352/10/LOT-10-ABC', 'LOT-10-ABC'],
  ]) {
    await enterBarcode(barcode);
    await page.getByRole('button', { name: 'Parse', exact: true }).click();
    await expect(lotField).toHaveValue(lot);
  }

  const observations = [];
  for (const lot of ['LOT-10-ABC', 'LOT-21-ABC', 'LOT-17260930']) {
    const barcode = `010950600013435210${lot}\x1d17260930`;
    await enterBarcode(barcode);
    await page.getByRole('button', { name: 'Parse', exact: true }).click();
    await expect(input).toHaveValue(''); // confirms React handler ran
    await expect(page.getByText('Raw Barcode String', { exact: true })).toBeVisible();
    observations.push({ barcode, expectedLot: lot, actualLot: await lotField.inputValue() });
  }
  expect(errors).toEqual([]);
  const screenshot = await writeRegressionEvidence(testInfo, 'fe-001', {
    url: page.url(), browser: testInfo.project.name, controlsPassed: 2,
    observations, pageErrors: errors, boundary: 'Real built scanner UI; native DOM setter/input event preserves ASCII GS fixture because Firefox fill strips it; no ingest or server persistence.',
  });
  await page.screenshot({ path: screenshot, fullPage: true });

  // Setup/controls/evidence are outside expected failure; only defect assertion is marked.
  test.fail(true, 'FE-001: variable-length GS1 lot is truncated at AI-like digits.');
  expect(observations.map(value => value.actualLot)).toEqual(observations.map(value => value.expectedLot));
});

The existing node-isolation.cjs protects file reads and off-host sockets.
Browser clients must intercept API requests and deny off-origin traffic.
"""
from pathlib import Path
import hashlib
import os
import secrets
import shutil
import subprocess
import tempfile
import time

ROOT = Path(__file__).resolve().parents[4]
OUT = ROOT / 'docs/qa/2026-10-02/artifacts/browser'
OUT.mkdir(parents=True, exist_ok=True)
BASE = Path(tempfile.mkdtemp(prefix='regengine-browser-followup-', dir='/private/tmp'))
APP = BASE / 'frontend'
APP.mkdir()
for name in ['next.config.js', 'package.json']:
    shutil.copy2(ROOT / 'frontend' / name, APP / name)
shutil.copytree(ROOT / 'frontend/public', APP / 'public')
shutil.copytree(ROOT / 'frontend/.next', APP / '.next',
                ignore=lambda directory, names: [name for name in names if name in {'cache', 'dev'}])
(APP / 'node_modules').symlink_to(ROOT / 'frontend/node_modules', target_is_directory=True)
GUARD = ROOT / 'docs/qa/2026-10-02/scripts/node-isolation.cjs'
env = {key: value for key, value in os.environ.items() if key in {'PATH', 'TMPDIR', 'LANG'}}
env.update({
    'NODE_OPTIONS': f'--require={GUARD}',
    'NEXT_TELEMETRY_DISABLED': '1', 'DO_NOT_TRACK': '1',
    'CSRF_SECRET': secrets.token_urlsafe(48),
    'ADMIN_SERVICE_URL': 'http://127.0.0.1:3113',
    'INGESTION_SERVICE_URL': 'http://127.0.0.1:3113',
    'COMPLIANCE_SERVICE_URL': 'http://127.0.0.1:3113',
    'NEXT_PUBLIC_ENABLE_EXPERIMENTAL_FEATURES': 'true',
})
command = ['node', str(ROOT / 'frontend/node_modules/next/dist/bin/next'),
           'start', '-H', '127.0.0.1', '-p', '3114']
process = subprocess.Popen(command, cwd=APP, env=env)
metadata = (f'PID={process.pid}\nPort=3114\nMirror={APP}\n'
            f'StopFile={BASE / "stop"}\nGuard SHA256={hashlib.sha256(GUARD.read_bytes()).hexdigest()}\n'
            f'Build ID={(APP / ".next/BUILD_ID").read_text().strip()}\n'
            'Runtime CSRF value: ephemeral synthetic, not recorded\n')
stamp = time.strftime('%Y%m%dT%H%M%SZ', time.gmtime())
(OUT / f'followup-server-{stamp}-metadata.txt').write_text(metadata)
print(metadata, flush=True)
try:
    deadline = time.monotonic() + 5100
    while process.poll() is None and time.monotonic() < deadline and not (BASE / 'stop').exists():
        time.sleep(.25)
finally:
    if process.poll() is None:
        process.terminate()
    try:
        process.wait(timeout=10)
    except subprocess.TimeoutExpired:
        process.kill()
        process.wait(timeout=5)
    print(f'QA built server stopped; exit {process.returncode}', flush=True)
