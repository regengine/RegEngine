from fastapi import FastAPI, HTTPException
from fastapi.testclient import TestClient
from shared.error_handling import ClientSafeValueError, install_exception_handlers


def _client() -> TestClient:
    app = FastAPI()

    @app.get("/http-unsafe-string")
    async def http_unsafe_string():
        raise HTTPException(
            status_code=500,
            detail="Traceback: connect to http://postgres.internal:5432 failed at /Users/sellers/app.py",
        )

    @app.get("/http-unsafe-dict")
    async def http_unsafe_dict():
        raise HTTPException(
            status_code=503,
            detail={
                "message": "Database http://postgres.internal unavailable",
                "dsn": "postgresql://[REDACTED]@postgres.internal/app",  # pragma: allowlist secret
            },
        )

    @app.get("/http-safe")
    async def http_safe():
        raise HTTPException(
            status_code=503,
            detail={"message": "Maintenance window active", "client_safe": True},
        )

    @app.get("/value-error")
    async def value_error():
        raise ValueError(
            "psycopg connect failed for postgresql://[REDACTED]@postgres.internal/app at /Users/sellers/app.py",  # pragma: allowlist secret
        )

    @app.get("/safe-value-error")
    async def safe_value_error():
        raise ClientSafeValueError("Invalid facility identifier")

    @app.get("/runtime-error")
    async def runtime_error():
        raise RuntimeError("stack frame at /Users/sellers/app.py for http://postgres.internal")

    install_exception_handlers(app)
    return TestClient(app, raise_server_exceptions=False)


def _assert_no_internal_details(body: dict) -> None:
    serialized = str(body)
    assert "postgres.internal" not in serialized
    assert "postgresql://" not in serialized
    assert "Traceback" not in serialized
    assert "/Users/sellers" not in serialized
    assert "psycopg" not in serialized


def test_5xx_http_exception_string_detail_is_sanitized() -> None:
    response = _client().get("/http-unsafe-string")

    assert response.status_code == 500
    body = response.json()
    assert body == {"error": {"type": "http_500", "message": "An internal error occurred"}}
    _assert_no_internal_details(body)


def test_5xx_http_exception_dict_detail_is_sanitized() -> None:
    response = _client().get("/http-unsafe-dict")

    assert response.status_code == 503
    body = response.json()
    assert body == {"error": {"type": "http_503", "message": "An internal error occurred"}}
    _assert_no_internal_details(body)


def test_client_safe_5xx_http_exception_message_can_be_exposed() -> None:
    response = _client().get("/http-safe")

    assert response.status_code == 503
    assert response.json() == {"error": {"type": "http_503", "message": "Maintenance window active"}}


def test_value_error_detail_is_sanitized_by_default() -> None:
    response = _client().get("/value-error")

    assert response.status_code == 400
    body = response.json()
    assert body == {"error": {"type": "bad_request", "message": "Invalid request"}}
    _assert_no_internal_details(body)


def test_client_safe_value_error_message_can_be_exposed() -> None:
    response = _client().get("/safe-value-error")

    assert response.status_code == 400
    assert response.json() == {"error": {"type": "bad_request", "message": "Invalid facility identifier"}}


def test_unhandled_exception_remains_generic() -> None:
    response = _client().get("/runtime-error")

    assert response.status_code == 500
    body = response.json()
    assert body["error"]["type"] == "internal_error"
    assert body["error"]["message"] == "An internal error occurred"
    _assert_no_internal_details(body)
