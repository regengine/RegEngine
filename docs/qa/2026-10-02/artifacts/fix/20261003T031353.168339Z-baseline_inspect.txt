def deny_external_io(event: str, args: tuple) -> None:
    if event in {"socket.connect", "socket.getaddrinfo", "subprocess.Popen", "os.system"}:
        raise RuntimeError(f"QA safety guard denied {event}; use an in-process mock")
    if event == "open" and isinstance(args[0], (str, bytes, os.PathLike)):
        path = Path(os.fsdecode(args[0]))
        if path.name.startswith(".env") or any(part in {".aws", ".ssh", ".gnupg"} for part in path.parts):
            raise RuntimeError("QA safety guard denied credential/environment file access")


sys.addaudithook(deny_external_io)


def deny_database_connect(*args, **kwargs):
    raise RuntimeError("QA safety guard denied database connection; use an in-process mock")


# Native DB drivers can bypass Python socket audit events. Block their public
# connection boundaries as well as SQLAlchemy's engine before test collection.
import importlib

from sqlalchemy.engine import Engine

Engine.connect = deny_database_connect
Engine.raw_connection = deny_database_connect
for driver in ("psycopg", "psycopg2", "asyncpg"):
    try:
        module = importlib.import_module(driver)
    except ImportError:
        continue
    module.connect = deny_database_connect
    if driver == "psycopg":
        module.Connection.connect = classmethod(deny_database_connect)
        module.AsyncConnection.connect = classmethod(deny_database_connect)

import sqlite3

_coverage_sqlite_connect = sqlite3.connect


def coverage_only_sqlite_connect(database, *args, **kwargs):
    path = Path(database).resolve()
    if path.parent != ROOT or not path.name.startswith(".coverage"):
        return deny_database_connect()
    return _coverage_sqlite_connect(database, *args, **kwargs)


sqlite3.connect = coverage_only_sqlite_connect
sqlite3.dbapi2.connect = coverage_only_sqlite_connect
